phase('Review')

  LENSES,
  (l) => agent(`你是评审，角度是「${l.name}」。评审的是「栖灯镇」的一段工作：${seg.title}。

本段要求：
${seg.brief}

${where}

实现的同事自述（不要轻信，以代码为准）：
${impl.summary}
取舍：${impl.decisions.join('；')}

${l.ask}

先看 --stat，再把和你这个角度有关的每个文件的差异读完（连同它调用到的现有代码），然后才下结论；covered 里逐条写你实际读过的文件和在里面核对了什么。改动不小，几分钟读不完，不要草草返回空数组。

只报你在代码里核实过、能给出具体触发场景的问题；风格和命名不算。只读，不要改工作树里的任何文件。没有问题就返回空数组。
${RULES}`, { label: `review:${seg.key}:${l.key}`, phase: 'Review', schema: FINDINGS }),
  (r, l) => {
    const fs = (r && r.findings) || []
    if (!fs.length) return { lens: l.key, items: [] }
    const indexed = fs.map((f, i) => ({ index: i, ...f }))
    return agent(`你是质疑者。下面是一位评审（角度「${l.name}」）对「${seg.title}」这段改动提的问题，每条带 index。你的任务是逐条尝试推翻：去读代码，${VERIFY_HOW}。拿不出证据证明它是真的，就判 real=false。除临时测试外不要改任何文件。

${where}

问题列表：
${JSON.stringify(indexed, null, 1)}

每条都要给出 verdict（index 对应）。
${RULES}`, { label: `verify:${seg.key}:${l.key}`, phase: 'Review', schema: VERDICTS })
      .then((v) => {
        const vs = (v && v.verdicts) || []
        return { lens: l.key, items: indexed.map((f) => ({ ...f, lens: l.key, verdict: vs.find((x) => x.index === f.index) || null })) }
      })
  },
)

const allItems = reviewed.filter(Boolean).flatMap((r) => r.items)
const confirmed = allItems.filter((f) => f.verdict && f.verdict.real)
const unverified = allItems.filter((f) => !f.verdict)
const rejected = allItems.filter((f) => f.verdict && !f.verdict.real)
log(`评审：共 ${allItems.length} 条，核实为真 ${confirmed.length} 条，被推翻 ${rejected.length} 条，未核实 ${unverified.length} 条`)

const FIX = {
  type: 'object',
  properties: {
    headCommit: { type: 'string' },
    fixed: { type: 'array', items: { type: 'string' }, description: '每条：问题标题 → 怎么修的、加了什么测试、反向验证结果' },
    notFixed: { type: 'array', items: { type: 'string' }, description: '没修的和理由' },
    checks: { type: 'array', items: { type: 'string' } },
  },
  required: ['headCommit', 'fixed', 'notFixed', 'checks'],
}

let fix = null
let recheck = null
const todo = [...confirmed, ...unverified]
if (todo.length) {
  phase('Fix')
  fix = await agent(`你来修「${seg.title}」这段改动里评审核实过的问题。工作地点是已有的工作树 ${impl.worktreePath}（分支 ${impl.branch}）：所有命令都先 cd 到那里，所有编辑都用那个目录下的绝对路径，不要在 /home/user/bot-game-mcp 主目录里改东西。

本段要求：
${seg.brief}

要处理的问题（verdict 为 null 的是没来得及核实的，你先自己核实，是真的再修）：
${JSON.stringify(todo, null, 1)}

每条按根因修，配一条会因为这个问题而失败的测试（先看到它红，再修到绿）。不要靠放宽或删除已有断言来通过。修完后把「完成的标准」里的检查全部重跑一遍（都用 flock）：${CHECKS}。在工作树分支上本地提交。
${RULES}`, { label: `fix:${seg.key}`, phase: 'Fix', schema: FIX })

  phase('Recheck')
  recheck = await agent(`你来复查一次修复。「${seg.title}」这段改动经过评审后做了一轮修复。工作树：${impl.worktreePath}（分支 ${impl.branch}）。修复之前的提交是 ${impl.headCommit}，看修复的改动：\`git -C ${impl.worktreePath} diff ${impl.headCommit}..HEAD\`；看整段改动：\`git -C ${impl.worktreePath} diff ${impl.baseCommit}..HEAD\`。

当初核实为真的问题：
${JSON.stringify(todo.map((f) => ({ title: f.title, file: f.file, scenario: f.scenario })), null, 1)}

修复的同事自述：
${fix ? JSON.stringify(fix, null, 1) : '（没有交回结果）'}

逐条判断：这个问题真的解决了吗（读代码，看对应测试是不是真的覆盖了那个场景）？修复有没有引入新的问题（尤其是隐私、事务、别的入口没同步）？有没有靠放宽断言解决的？只读，不要改文件。
${RULES}`, {
    label: `recheck:${seg.key}`, phase: 'Recheck',
    schema: {
      type: 'object',
      properties: {
        unresolved: { type: 'array', items: { type: 'string' } },
        newIssues: { type: 'array', items: { type: 'string' } },
        ok: { type: 'boolean' },
        notes: { type: 'string' },
      },
      required: ['unresolved', 'newIssues', 'ok'],
    },
  })
}

return {
  segment: seg.key,
  impl,
  counts: { total: allItems.length, confirmed: confirmed.length, rejected: rejected.length, unverified: unverified.length },
  confirmed: confirmed.map((f) => ({ title: f.title, file: f.file, severity: f.severity, lens: f.lens, scenario: f.scenario })),
  rejected: rejected.map((f) => ({ title: f.title, why: f.verdict.evidence })),
  fix,
  recheck,
}
